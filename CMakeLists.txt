cmake_minimum_required(VERSION 3.24)
project(windows-memory-pressure-explainer VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ENABLE_ANALYZE "Run MSVC code analysis" OFF)
add_library(core src/core.cpp)
target_include_directories(core PUBLIC src)
add_executable(windows-memory-pressure-explainer src/main.cpp src/windows.cpp)
target_link_libraries(windows-memory-pressure-explainer PRIVATE core psapi)
add_executable(unit-tests tests/tests.cpp)
target_link_libraries(unit-tests PRIVATE core)
foreach(t core windows-memory-pressure-explainer unit-tests)
 target_compile_options(${t} PRIVATE /W4 /WX /utf-8)
 target_compile_definitions(${t} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 if(ENABLE_ANALYZE)
  target_compile_options(${t} PRIVATE /analyze)
 endif()
endforeach()
enable_testing()
add_test(NAME unit COMMAND unit-tests)
add_test(NAME help COMMAND windows-memory-pressure-explainer --help)
add_test(NAME version COMMAND windows-memory-pressure-explainer --version)
add_executable(json-fixture tests/fixture.cpp)
target_link_libraries(json-fixture PRIVATE core)
target_compile_options(json-fixture PRIVATE /W4 /WX /utf-8)
if(ENABLE_ANALYZE)
 target_compile_options(json-fixture PRIVATE /analyze)
endif()
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
add_test(NAME json-contract COMMAND ${POWERSHELL_EXECUTABLE} -NoProfile -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/json-test.ps1 -Fixture $<TARGET_FILE:json-fixture>)
enable_language(RC)
target_sources(windows-memory-pressure-explainer PRIVATE src/version.rc)
if(MSVC)
 target_link_options(windows-memory-pressure-explainer PRIVATE /MANIFEST:EMBED /MANIFESTUAC:level='asInvoker')
endif()
